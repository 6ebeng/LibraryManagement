\subsection{Scope}
The Static Testing covers:
\begin{itemize}
    \item Code reviews and inspections
    \item Walkthroughs of critical components
    \item Documentation reviews (SRS, design documents, test plans)
    \item Static analysis tool integration
    \item Coding standards compliance
    \item Security vulnerability analysis
    \item Performance and maintainability assessment
\end{itemize}

\subsection{Static Testing Types}
\begin{enumerate}
    \item \textbf{Informal Reviews}: Buddy checks and peer reviews
    \item \textbf{Formal Reviews}: Structured inspections with defined roles
    \item \textbf{Walkthroughs}: Author-led presentations of code/documents
    \item \textbf{Technical Reviews}: Focus on technical accuracy and standards
    \item \textbf{Automated Static Analysis}: Tool-based code analysis
\end{enumerate}

\hline
\textbf{Functionality} & 
\begin{itemize}[nosep]
    \item Code implements requirements correctly
    \item Business logic is accurate
    \item Edge cases are handled
    \item Error conditions are properly managed
    \item Input validation is comprehensive

\begin{itemize}[nosep]

    \item Variable and function names are descriptive
    \item Code follows DRY principle

\hline

\begin{itemize}[nosep]
    \item Input sanitization is implemented
    \item SQL injection prevention measures
    \item XSS protection mechanisms
    \item Authentication and authorization checks

\end{itemize} \\
\hline

    \item Memory usage is reasonable
    \item No unnecessary loops or operations
    \item Caching is implemented where appropriate
\end{itemize} \\
\hline

\begin{itemize}[nosep]

    \item Comments explain complex logic

\end{itemize} \\

    \item Test cases cover edge cases

\hline
\rowcolor{lightgray}
\textbf{Standard Category} & \textbf{Compliance Criteria} \\
\hline
\textbf{Naming Conventions} & 
\begin{itemize}[nosep]
    \item Variables use camelCase
    \item Constants use UPPER\_SNAKE\_CASE
    \item Functions use descriptive names
    \item Classes use PascalCase
    \item Files use kebab-case or camelCase consistently

\hline
\textbf{Code Structure} & 
\begin{itemize}[nosep]
    \item Consistent indentation (2 or 4 spaces)
    \item Maximum line length (80-120 characters)
    \item Proper use of whitespace
    \item Logical code organization
    \item Consistent bracket placement
\end{itemize} \\
\hline
\textbf{Documentation} & 
\begin{itemize}[nosep]
    \item JSDoc comments for functions
    \item Inline comments for complex logic
    \item README files for modules
    \item API documentation maintained
    \item Change logs updated
\end{itemize} \\
\hline
\textbf{Error Handling} & 
\begin{itemize}[nosep]
    \item Consistent error handling patterns
    \item Proper use of try-catch blocks
    \item Meaningful error messages
    \item Error logging implemented
    \item Graceful degradation
\end{itemize} \\
\hline
\textbf{Testing Standards} & 
\begin{itemize}[nosep]
    \item Test files follow naming convention
    \item Test descriptions are clear
    \item Arrange-Act-Assert pattern used
    \item Mocking is used appropriately
    \item Test data is isolated
\end{itemize} \\
\hline
\end{longtable}
