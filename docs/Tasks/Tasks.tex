\section*{Work Breakdown Structure - Phases}

\begin{itemize}[leftmargin=*] % Main Phases as an unnumbered list
    \item \stepcounter{phasecounter}\textbf{Phase \thephasecounter: Planning \& Initial Static Testing}
        \begin{enumerate}[label=\thephasecounter.\arabic*., labelindent=1em, leftmargin=*] % Sub-tasks numbered with phase
            \item Understand SRS \& Coursework Requirements
            \item Develop Detailed Test Plan (incl. techniques, criteria) \& Task Distribution
            \item VIII. Static Testing: Documentation Review (SRS, Test Plan)
            \item VIII. Static Testing: Initial Code Review Setup \& Process Definition
            \item \textbf{Milestone:} P1 Complete: Test Plan Finalized
        \end{enumerate}

    \item \stepcounter{phasecounter}\textbf{Phase \thephasecounter: Core Functional \& API Test Design \& Execution}
        \begin{enumerate}[label=\thephasecounter.\arabic*., labelindent=1em, leftmargin=*]
            \item I.A Functional: Authentication Test Design (Non-Coders)
            \item I.B Functional: Entity Mgt. (incl. Data Integrity \& Constraint) Test Design (Non-Coders)
            \item I.C-D Functional: Feature \& Use Case (incl. Error Handling) Test Design (Non-Coders)
            \item I.E Functional: State Transition Test Design (Non-Coders)
            \item II. API Test Scenario Design (Non-Coders)
            \item II. API Test Automation Setup \& Initial Scripts (Coders)
            \item I. Functional Test Execution \& Defect Reporting (Non-Coders)
            \item II. API Test Execution (Manual \& Automated) (Team)
            \item \textbf{Milestone:} P2 Complete: Core Functional/API Tests Executed
        \end{enumerate}

    \item \stepcounter{phasecounter}\textbf{Phase \thephasecounter: White-Box \& Non-Functional Testing}
        \begin{enumerate}[label=\thephasecounter.\arabic*., labelindent=1em, leftmargin=*]
            \item III. White-Box Test Planning \& Unit Test Development (Coders)
            \item III. White-Box Test Execution \& Code Coverage Analysis (e.g., Decision, Path) (Coders)
            \item IV.A Security Test Design \& Execution (Team)
            \item IV.B Usability Test Design \& Execution (Non-Coders)
            \item IV.C Performance (incl. Basic Network Conditions) Test Design \& Execution (Team)
            \item IV.E Browser Compatibility Testing (Non-Coders)
            \item VIII. Static Testing: Ongoing Code Reviews (Team)
            \item \textbf{Milestone:} P3 Complete: White-Box/NFR Tests Executed
        \end{enumerate}

    \item \stepcounter{phasecounter}\textbf{Phase \thephasecounter: Integration \& Early Regression Testing}
        \begin{enumerate}[label=\thephasecounter.\arabic*., labelindent=1em, leftmargin=*]
            \item V. Integration Test Scenario Design (Non-Coders)
            \item V. Integration Test Execution (Team)
            \item VI. Regression Test Suite Development (Manual - Non-Coders)
            \item VI. Regression Test Suite Development (Automated - Coders)
            \item VI. Early Regression Test Cycles (Team)
            \item \textbf{Milestone:} P4 Complete: Integration \& Initial Regression Done
        \end{enumerate}

    \item \stepcounter{phasecounter}\textbf{Phase \thephasecounter: System Changes, Final Testing \& Report}
        \begin{enumerate}[label=\thephasecounter.\arabic*., labelindent=1em, leftmargin=*]
            \item Suggest, Implement \& Test System Changes (Team)
            \item VI. Final Regression Testing Cycles (Team)
            \item VII. Installation/Deployment Testing (Team)
            \item VIII. Static Testing: Final Documentation Review (Team)
            \item Prepare Final Report \& Presentation (Team)
            \item \textbf{Milestone:} Project Submission
        \end{enumerate}
\end{itemize}
